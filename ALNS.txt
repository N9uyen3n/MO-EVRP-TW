// ============================================================================
// CONSTRUCTOR - Enhanced with better defaults
// ============================================================================
ALNS::ALNS(ObjectiveManager manager, std::shared_ptr<Instance> instance) 
    : instance(std::move(instance)),
      objectiveManager(std::move(manager)), 
      rng(std::random_device{}())
{
    if (!this->instance) {
        throw std::runtime_error("Instance cannot be null in ALNS constructor");
    }

    // Enhanced configuration
    maxIterations = 5000;
    segmentSize = 100;
    destructionRate = 0.3; // Increased for more exploration
    
    temperature = 150.0;
    coolingRate = 0.9975;
    minimumTemperature = 0.1;
    initialTemperature = temperature;
    
    reactionFactor = 0.15;

    // Improved reward structure
    sigma1 = 50;  // New global best - highest reward
    sigma2 = 25;  // Better than current
    sigma3 = 10;  // Accepted (non-dominated)

    initializeOperators();
    
    std::cout << "ALNS initialized with " << destroyOperators.size() 
              << " destroy and " << repairOperators.size() 
              << " repair operators." << std::endl;
}

// ============================================================================
// OPERATOR INITIALIZATION
// ============================================================================
void ALNS::initializeOperators() {
    destroyOperators.push_back(std::make_unique<RandomRemoval>());
    destroyOperators.push_back(std::make_unique<WorstDistanceRemoval>());
    destroyOperators.push_back(std::make_unique<ShawRemoval>());
    destroyOperators.push_back(std::make_unique<ZoneRemoval>());

    repairOperators.push_back(std::make_unique<GreedyInsertion>());
    repairOperators.push_back(std::make_unique<RegretInsertion>());

    const size_t numDestroy = destroyOperators.size();
    const size_t numRepair = repairOperators.size();
    
    destroyWeights.assign(numDestroy, 1.0);
    repairWeights.assign(numRepair, 1.0);
    destroyScores.assign(numDestroy, 0);
    repairScores.assign(numRepair, 0);
    destroyUses.assign(numDestroy, 0);
    repairUses.assign(numRepair, 0);
    destroyBestScores.assign(numDestroy, 0);
    repairBestScores.assign(numRepair, 0);
}

// ============================================================================
// IMPROVED LOCAL SEARCH - Multiple Operators
// ============================================================================
void ALNS::applyLocalSearch(Solution& solution) {
    bool globalImprovement = true;
    int maxIterations = 3; // Limit LS iterations to avoid excessive time
    int iteration = 0;
    
    while (globalImprovement && iteration < maxIterations) {
        globalImprovement = false;
        iteration++;
        
        // 1. Inter-Route Relocate
        if (applyRelocate(solution)) {
            globalImprovement = true;
            continue;
        }
        
        // 2. Inter-Route Exchange (2-opt*)
        if (applyExchange(solution)) {
            globalImprovement = true;
            continue;
        }
        
        // 3. Intra-Route 2-opt
        if (applyIntraRouteOptimization(solution)) {
            globalImprovement = true;
            continue;
        }
        
        // 4. Station Optimization
        if (optimizeStations(solution)) {
            globalImprovement = true;
        }
    }
    
    // Clean up empty routes
    solution.routes.erase(
        std::remove_if(solution.routes.begin(), solution.routes.end(),
            [](const Route& r) { return r.getInfos().size() <= 2; }),
        solution.routes.end()
    );
}

// --- Relocate: Move customer from one route to another ---
bool ALNS::applyRelocate(Solution& solution) {
    for (size_t i = 0; i < solution.routes.size(); ++i) {
        for (size_t j = 0; j < solution.routes.size(); ++j) {
            if (i == j) continue;

            Route& route_i = solution.routes[i];
            Route& route_j = solution.routes[j];

            for (size_t c_idx = 1; c_idx < route_i.getInfos().size() - 1; ++c_idx) {
                auto customerToMove = std::dynamic_pointer_cast<Customer>(
                    route_i.getInfos()[c_idx].node);
                if (!customerToMove) continue;

                // Calculate removal cost
                auto prevNode = route_i.getInfos()[c_idx - 1].node;
                auto nextNode = route_i.getInfos()[c_idx + 1].node;
                double cost_i_old = instance->getDistance(prevNode->getId(), customerToMove->getId()) +
                                  instance->getDistance(customerToMove->getId(), nextNode->getId());
                double cost_i_new = instance->getDistance(prevNode->getId(), nextNode->getId());
                double costSaving = cost_i_old - cost_i_new;

                int bestNewPos = -1;
                double bestTotalDelta = 1e-6; // Small positive threshold

                // Find best insertion position
                for (size_t pos_j = 1; pos_j < route_j.getInfos().size(); ++pos_j) {
                    if (route_j.canInsert(customerToMove, pos_j)) {
                        double costIncrease = route_j.getInsertionCost(customerToMove, pos_j);
                        double totalDelta = costIncrease - costSaving;

                        if (totalDelta < bestTotalDelta) {
                            bestTotalDelta = totalDelta;
                            bestNewPos = pos_j;
                        }
                    }
                }

                if (bestNewPos != -1) {
                    route_i.removeCustomer(customerToMove);
                    route_j.insert(customerToMove, bestNewPos);
                    return true; // Return immediately after first improvement
                }
            }
        }
    }
    return false;
}

// --- Exchange: Swap customers between two routes ---
bool ALNS::applyExchange(Solution& solution) {
    for (size_t i = 0; i < solution.routes.size(); ++i) {
        for (size_t j = i + 1; j < solution.routes.size(); ++j) {
            Route& route_i = solution.routes[i];
            Route& route_j = solution.routes[j];

            for (size_t ci = 1; ci < route_i.getInfos().size() - 1; ++ci) {
                auto cust_i = std::dynamic_pointer_cast<Customer>(route_i.getInfos()[ci].node);
                if (!cust_i) continue;

                for (size_t cj = 1; cj < route_j.getInfos().size() - 1; ++cj) {
                    auto cust_j = std::dynamic_pointer_cast<Customer>(route_j.getInfos()[cj].node);
                    if (!cust_j) continue;

                    // Try swapping customers
                    if (canSwap(route_i, ci, route_j, cj)) {
                        double deltaCost = calculateSwapCost(route_i, ci, route_j, cj);
                        
                        if (deltaCost < -1e-6) {
                            performSwap(route_i, ci, route_j, cj);
                            return true;
                        }
                    }
                }
            }
        }
    }
    return false;
}

// --- Intra-Route 2-opt: Reverse segment within a route ---
bool ALNS::applyIntraRouteOptimization(Solution& solution) {
    for (auto& route : solution.routes) {
        auto& infos = route.getInfos();
        
        for (size_t i = 1; i < infos.size() - 2; ++i) {
            for (size_t j = i + 1; j < infos.size() - 1; ++j) {
                // Calculate cost improvement
                double oldCost = instance->getDistance(infos[i-1].node->getId(), infos[i].node->getId()) +
                               instance->getDistance(infos[j].node->getId(), infos[j+1].node->getId());
                double newCost = instance->getDistance(infos[i-1].node->getId(), infos[j].node->getId()) +
                               instance->getDistance(infos[i].node->getId(), infos[j+1].node->getId());
                
                if (newCost < oldCost - 1e-6) {
                    // Try reversing segment [i, j]
                    if (canReverse(route, i, j)) {
                        reverseSegment(route, i, j);
                        return true;
                    }
                }
            }
        }
    }
    return false;
}

// --- Station Optimization: Add/Remove charging stations ---
bool ALNS::optimizeStations(Solution& solution) {
    bool improved = false;
    
    for (auto& route : solution.routes) {
        // Try removing unnecessary stations
        improved |= removeUnnecessaryStations(route);
        
        // Try adding stations where needed
        improved |= addNecessaryStations(route);
    }
    
    return improved;
}

// Helper: Check if two customers can be swapped
bool ALNS::canSwap(Route& route_i, size_t pos_i, Route& route_j, size_t pos_j) {
    auto cust_i = std::dynamic_pointer_cast<Customer>(route_i.getInfos()[pos_i].node);
    auto cust_j = std::dynamic_pointer_cast<Customer>(route_j.getInfos()[pos_j].node);
    
    if (!cust_i || !cust_j) return false;
    
    // Check capacity constraints
    double demand_i = cust_i->getDemand();
    double demand_j = cust_j->getDemand();
    
    // Calculate current loads
    double load_i = 0, load_j = 0;
    for (const auto& info : route_i.getInfos()) {
        if (auto c = std::dynamic_pointer_cast<Customer>(info.node)) {
            load_i += c->getDemand();
        }
    }
    for (const auto& info : route_j.getInfos()) {
        if (auto c = std::dynamic_pointer_cast<Customer>(info.node)) {
            load_j += c->getDemand();
        }
    }
    
    double new_load_i = load_i - demand_i + demand_j;
    double new_load_j = load_j - demand_j + demand_i;
    
    if (new_load_i > route_i.getVehicle()->getCapacity() || 
        new_load_j > route_j.getVehicle()->getCapacity()) {
        return false;
    }
    
    return true;
}

// Helper: Calculate cost delta of swapping
double ALNS::calculateSwapCost(Route& route_i, size_t pos_i, Route& route_j, size_t pos_j) {
    const auto& infos_i = route_i.getInfos();
    const auto& infos_j = route_j.getInfos();
    
    // Old costs
    double old_cost_i = instance->getDistance(infos_i[pos_i-1].node->getId(), infos_i[pos_i].node->getId()) +
                       instance->getDistance(infos_i[pos_i].node->getId(), infos_i[pos_i+1].node->getId());
    double old_cost_j = instance->getDistance(infos_j[pos_j-1].node->getId(), infos_j[pos_j].node->getId()) +
                       instance->getDistance(infos_j[pos_j].node->getId(), infos_j[pos_j+1].node->getId());
    
    // New costs (after swap)
    double new_cost_i = instance->getDistance(infos_i[pos_i-1].node->getId(), infos_j[pos_j].node->getId()) +
                       instance->getDistance(infos_j[pos_j].node->getId(), infos_i[pos_i+1].node->getId());
    double new_cost_j = instance->getDistance(infos_j[pos_j-1].node->getId(), infos_i[pos_i].node->getId()) +
                       instance->getDistance(infos_i[pos_i].node->getId(), infos_j[pos_j+1].node->getId());
    
    return (new_cost_i + new_cost_j) - (old_cost_i + old_cost_j);
}

// Helper: Perform the swap
void ALNS::performSwap(Route& route_i, size_t pos_i, Route& route_j, size_t pos_j) {
    auto temp = route_i.getInfos()[pos_i].node;
    route_i.getInfos()[pos_i].node = route_j.getInfos()[pos_j].node;
    route_j.getInfos()[pos_j].node = temp;
    
    route_i.recalculateFrom(0);
    route_j.recalculateFrom(0);
}

    // Simple heuristic: check if reversal doesn't violate time windows drastically

// Helper: Reverse a segment
void ALNS::reverseSegment(Route& route, size_t start, size_t end) {
    auto& infos = route.getInfos();
    std::reverse(infos.begin() + start, infos.begin() + end + 1);
    route.recalculateFrom(0);
}

// Helper: Remove unnecessary charging stations
bool ALNS::removeUnnecessaryStations(Route& route) {
    bool removed = false;
    auto& infos = route.getInfos();
    
    for (size_t i = 1; i < infos.size() - 1; ++i) {
        if (auto station = std::dynamic_pointer_cast<Station>(infos[i].node)) {
            // Try removing this station
            route.remove(i);
            
            // Check if route is still feasible
            if (isFeasible(Solution{{route}})) {
                removed = true;
                break; // Remove one at a time
            } else {
                // Reinsert if not feasible
                route.insert(station, i);
            }
        }
    }
    
    return removed;
}

// Helper: Add necessary charging stations
bool ALNS::addNecessaryStations(Route& route) {
    // Find positions where battery runs low
    const auto& infos = route.getInfos();
    
    for (size_t i = 1; i < infos.size(); ++i) {
        if (infos[i].arrival_battery < 10.0) { // Low battery threshold
            // Find nearest station and try to insert before this position
            auto nearestStation = findNearestStation(infos[i-1].node);
            
            if (nearestStation && route.canInsert(nearestStation, i)) {
                route.insert(nearestStation, i);
                return true;
            }
        }
    }
    
    return false;
}

// Helper: Find nearest charging station
std::shared_ptr<Station> ALNS::findNearestStation(std::shared_ptr<Node> from) {
    std::shared_ptr<Station> nearest = nullptr;
    double minDist = std::numeric_limits<double>::max();
    
    for (const auto& node : instance->getNodes()) {
        if (auto station = std::dynamic_pointer_cast<Station>(node)) {
            double dist = instance->getDistance(from->getId(), station->getId());
            if (dist < minDist) {
                minDist = dist;
                nearest = station;
            }
        }
    }
    
    return nearest;
}

// ============================================================================
// IMPROVED FEASIBILITY CHECK
// ============================================================================
bool ALNS::isFeasible(const Solution& solution) {
    if (solution.routes.empty()) {
        return true;
    }
    
    constexpr double EPSILON = 1e-4;
    std::set<int> customers_served;
    int total_customers_in_solution = 0;

    for (const auto& route : solution.routes) {
        const auto& infos = route.getInfos();
        
        if (infos.empty()) continue;
        
        if (infos.size() < 2) return false;
        
        if (!std::dynamic_pointer_cast<Depot>(infos.front().node) ||
            !std::dynamic_pointer_cast<Depot>(infos.back().node)) {
            return false;
        }

        auto vehicle = route.getVehicle();
        const double capacity = vehicle->getCapacity();
        const double batteryCapacity = vehicle->getBatteryCapacity();

        for (const auto& info : infos) {
            // Battery constraints
            if (info.arrival_battery < -EPSILON || 
                info.departure_battery > batteryCapacity + EPSILON) {
                return false;
            }

            // Load constraints
            if (info.departure_load < -EPSILON || 
                info.departure_load > capacity + EPSILON) {
                return false;
            }

            // Time window constraints
            if (auto customer = std::dynamic_pointer_cast<const Customer>(info.node)) {
                if (info.arrival_time > customer->getDueDate() + EPSILON) {
                    return false;
                }
                
                if (customers_served.count(customer->getId())) {
                    return false;
                }
                customers_served.insert(customer->getId());
                total_customers_in_solution++;
            }
        }
    }

    // Check all customers are served
    int total_customers_in_instance = 0;
    for (const auto& node : instance->getNodes()) {
        if (std::dynamic_pointer_cast<Customer>(node)) {
            total_customers_in_instance++;
        }
    }
    
    return total_customers_in_solution == total_customers_in_instance;
}

// ============================================================================
// IMPROVED INITIAL SOLUTION - Clarke-Wright Savings
// ============================================================================
Solution ALNS::generateInitialSolution() {
    std::cout << "Generating initial solution using Clarke-Wright..." << std::endl;
    
    Solution sol;
    std::shared_ptr<Node> depot = nullptr;
    std::vector<std::shared_ptr<Customer>> customers;

    for (const auto& node : instance->getNodes()) {
        if (dynamic_cast<Depot*>(node.get())) {
            depot = node;
        } else if (auto customer = std::dynamic_pointer_cast<Customer>(node)) {
            customers.push_back(customer);
        }
    }

    if (!depot) {
        throw std::runtime_error("Depot not found!");
    }

    // Calculate savings for all customer pairs
    struct Saving {
        std::shared_ptr<Customer> i;
        std::shared_ptr<Customer> j;
        double value;
        
        bool operator<(const Saving& other) const {
            return value > other.value; // Descending order
        }
    };
    
    std::vector<Saving> savings;
    for (size_t i = 0; i < customers.size(); ++i) {
        for (size_t j = i + 1; j < customers.size(); ++j) {
            double saving = instance->getDistance(depot->getId(), customers[i]->getId()) +
                           instance->getDistance(depot->getId(), customers[j]->getId()) -
                           instance->getDistance(customers[i]->getId(), customers[j]->getId());
            savings.push_back({customers[i], customers[j], saving});
        }
    }
    
    std::sort(savings.begin(), savings.end());
    
    // Create initial routes (one per customer)
    int routeId = 0;
    std::map<std::shared_ptr<Customer>, size_t> customerToRoute;
    
    for (const auto& customer : customers) {
        auto vehicle = std::make_shared<Vehicle>(
            routeId, 
            instance->getVehicleCapacity(), 
            instance->getVehicleBattery(), 
            instance->getVehicleEnergyRate()
        );
        
        Route newRoute(routeId++, vehicle, instance);
        newRoute.insert(depot, 0);
        newRoute.insert(customer, 1);
        newRoute.insert(depot, 2);
        
        customerToRoute[customer] = sol.routes.size();
        sol.routes.push_back(newRoute);
    }
    
    // Merge routes based on savings
    for (const auto& saving : savings) {
        size_t route_i = customerToRoute[saving.i];
        size_t route_j = customerToRoute[saving.j];
        
        if (route_i == route_j) continue; // Already in same route
        
        // Try to merge routes
        if (canMergeRoutes(sol.routes[route_i], sol.routes[route_j])) {
            mergeRoutes(sol.routes[route_i], sol.routes[route_j]);
            
            // Update customer mapping
            for (auto& pair : customerToRoute) {
                if (pair.second == route_j) {
                    pair.second = route_i;
                }
            }
        }
    }
    
    // Remove empty routes
    sol.routes.erase(
        std::remove_if(sol.routes.begin(), sol.routes.end(),
            [](const Route& r) { return r.getInfos().size() <= 2; }),
        sol.routes.end()
    );
    
    std::cout << "Initial solution: " << sol.routes.size() << " routes" << std::endl;
    return sol;
}

// Helper: Check if two routes can be merged
bool ALNS::canMergeRoutes(const Route& r1, const Route& r2) {
    double total_demand = 0;
    
    for (const auto& info : r1.getInfos()) {
        if (auto c = std::dynamic_pointer_cast<Customer>(info.node)) {
            total_demand += c->getDemand();
        }
    }
    
    for (const auto& info : r2.getInfos()) {
        if (auto c = std::dynamic_pointer_cast<Customer>(info.node)) {
            total_demand += c->getDemand();
        }
    }
    
    return total_demand <= instance->getVehicleCapacity();
}

// Helper: Merge two routes
void ALNS::mergeRoutes(Route& r1, Route& r2) {
    // Insert all customers from r2 into r1
    const auto& infos_r2 = r2.getInfos();
    
    for (size_t i = 1; i < infos_r2.size() - 1; ++i) {
        r1.insert(infos_r2[i].node, r1.getInfos().size() - 1);
    }
    
    // Clear r2
    while (r2.getInfos().size() > 2) {
        r2.remove(1);
    }
}

// ============================================================================
// MAIN SOLVE METHOD - Enhanced
// ============================================================================
std::vector<Solution> ALNS::solve() {
    if (!instance) {
        throw std::runtime_error("Instance is null");
    }

    auto startTime = std::chrono::high_resolution_clock::now();
    printHeader();
    
    Solution currentSolution = generateInitialSolution();
    calculateObjectives(currentSolution);
    updateArchive(currentSolution);
    
    Solution bestSolution = currentSolution;
    int iterationsSinceImprovement = 0;
    int acceptedSolutions = 0;
    int rejectedSolutions = 0;
    int totalFeasible = 0;
    int totalInfeasible = 0;

    for (int iteration = 0; iteration < maxIterations; ++iteration) {
        Solution tempSolution = currentSolution;

        auto destroyPair = selectOperator(destroyOperators, destroyWeights);
        IDestroyOperator* destroyOp = destroyPair.first;
        size_t destroyIdx = destroyPair.second;

        auto repairPair = selectOperator(repairOperators, repairWeights);
        IRepairOperator* repairOp = repairPair.first;
        size_t repairIdx = repairPair.second;
        
        destroyUses[destroyIdx]++;
        repairUses[repairIdx]++;

        auto removedCustomers = destroyOp->destroy(tempSolution, *instance, destructionRate, rng);
        bool repaired = repairOp->repair(tempSolution, *instance, removedCustomers, rng);

        if (repaired) {
            applyLocalSearch(tempSolution);
        }

        if (repaired && isFeasible(tempSolution)) {
            totalFeasible++;
            calculateObjectives(tempSolution);

            Dominance status = dominanceCheck(tempSolution, currentSolution);
            bool addedToArchive = updateArchive(tempSolution);

            int reward = 0;
            if (addedToArchive) {
                reward = sigma1;
                iterationsSinceImprovement = 0;
                
                if (dominanceCheck(tempSolution, bestSolution) == DOMINATES) {
                    bestSolution = tempSolution;
                }
            } else if (status == DOMINATES) {
                reward = sigma2;
                iterationsSinceImprovement = 0;
            } else if (status != DOMINATED) {
                reward = sigma3;
            }

            if (reward > 0) {
                destroyScores[destroyIdx] += reward;
                repairScores[repairIdx] += reward;
                destroyBestScores[destroyIdx] = std::max(destroyBestScores[destroyIdx], static_cast<size_t>(reward));
                repairBestScores[repairIdx] = std::max(repairBestScores[repairIdx], static_cast<size_t>(reward));
            }

            bool accepted = false;
            if (status == DOMINATES) {
                currentSolution = tempSolution;
                accepted = true;
            } else if (status != DOMINATED) {
                double acceptProb = std::exp(-1.0 / temperature);
                std::uniform_real_distribution<> dist(0.0, 1.0);
                if (dist(rng) < acceptProb) {
                    currentSolution = tempSolution;
                    accepted = true;
                }
            }

            if (accepted) {
                acceptedSolutions++;
            } else {
                rejectedSolutions++;
            }
        } else {
            totalInfeasible++;
        }

        if (iteration > 0 && iteration % segmentSize == 0) {
            updateWeights();
            
            if (iteration % (segmentSize * 5) == 0) {
                logProgress(iteration, acceptedSolutions, rejectedSolutions, 
                           totalFeasible, totalInfeasible);
            }
        }

        temperature *= coolingRate;
        if (temperature < minimumTemperature) {
            temperature = minimumTemperature;
        }
        
        if (iterationsSinceImprovement > maxIterations / 4) {
            temperature = initialTemperature * 0.5;
            iterationsSinceImprovement = 0;
            std::cout << "  [Reheating] Temperature reset" << std::endl;
        }
        
        iterationsSinceImprovement++;
    }

    auto endTime = std::chrono::high_resolution_clock::now();
    auto duration = std::chrono::duration_cast<std::chrono::milliseconds>(endTime - startTime);
    
    printFooter(duration.count(), acceptedSolutions, rejectedSolutions, 
                totalFeasible, totalInfeasible);
    
    return archive;
}

// ============================================================================
// REMAINING METHODS (unchanged from original)
// ============================================================================
void ALNS::calculateObjectives(Solution& solution) {
    std::vector<double> values = objectiveManager.evaluateAll(solution);
    solution.setObjectiveValues(values);
}

std::pair<IDestroyOperator*, size_t> ALNS::selectOperator(
    const std::vector<std::unique_ptr<IDestroyOperator>>& operators, 
    const std::vector<double>& weights) 
{
    std::vector<double> positiveWeights = weights;
    for (auto& w : positiveWeights) {
        w = std::max(w, 0.01);
    }
    
    std::discrete_distribution<> dist(positiveWeights.begin(), positiveWeights.end());
    size_t index = dist(rng);
    return {operators[index].get(), index};
}

std::pair<IRepairOperator*, size_t> ALNS::selectOperator(
    const std::vector<std::unique_ptr<IRepairOperator>>& operators, 
    const std::vector<double>& weights) 
{
    std::vector<double> positiveWeights = weights;
    for (auto& w : positiveWeights) {
        w = std::max(w, 0.01);
    }
    
    std::discrete_distribution<> dist(positiveWeights.begin(), positiveWeights.end());
    size_t index = dist(rng);
    return {operators[index].get(), index};
}

void ALNS::updateWeights() {
    for (size_t i = 0; i < destroyOperators.size(); ++i) {
        if (destroyUses[i] > 0) {
            double avgScore = static_cast<double>(destroyScores[i]) / destroyUses[i];
            destroyWeights[i] = (1.0 - reactionFactor) * destroyWeights[i] + 
                               reactionFactor * avgScore;
            destroyWeights[i] = std::max(destroyWeights[i], 0.1);
        }
    }

    for (size_t i = 0; i < repairOperators.size(); ++i) {
        if (repairUses[i] > 0) {
            double avgScore = static_cast<double>(repairScores[i]) / repairUses[i];
            repairWeights[i] = (1.0 - reactionFactor) * repairWeights[i] + 
                              reactionFactor * avgScore;
            repairWeights[i] = std::max(repairWeights[i], 0.1);
        }
    }

    std::fill(destroyScores.begin(), destroyScores.end(), 0);
    std::fill(repairScores.begin(), repairScores.end(), 0);
    std::fill(destroyUses.begin(), destroyUses.end(), 0);
    std::fill(repairUses.begin(), repairUses.end(), 0);
}

bool ALNS::updateArchive(Solution& newSolution) {
    for (const auto& existing : archive) {
        if (dominanceCheck(newSolution, existing) == DOMINATED) {
            return false;
        }
    }

    size_t initialSize = archive.size();
    archive.erase(
        std::remove_if(archive.begin(), archive.end(),
            [&](const Solution& existing) {
                return dominanceCheck(newSolution, existing) == DOMINATES;
            }),
        archive.end()
    );

    archive.push_back(newSolution);
    return archive.size() < initialSize + 1;
}

ALNS::Dominance ALNS::dominanceCheck(const Solution& a, const Solution& b) {
    const auto& objA = a.objectives;
    const auto& objB = b.objectives;

    if (objA.size() != objB.size()) {
        throw std::runtime_error("Objective vectors have different sizes");
    }

    bool aBetter = false;
    bool bBetter = false;

    for (size_t i = 0; i < objA.size(); ++i) {
        if (objA[i] < objB[i] - 1e-6) {
            aBetter = true;
        } else if (objB[i] < objA[i] - 1e-6) {
            bBetter = true;
        }
        
        if (aBetter && bBetter) {
            return NON_DOMINATED;
        }
    }

    if (aBetter && !bBetter) return DOMINATES;
    if (bBetter && !aBetter) return DOMINATED;
    return NON_DOMINATED;
}

void ALNS::printFooter(long long duration, int accepted, int rejected,
                       int feasible, int infeasible) const {
    std::cout << std::string(80, '=') << std::endl;
    std::cout << "                    ALNS SOLVER FINISHED" << std::endl;
    std::cout << std::string(80, '=') << std::endl;
    std::cout << "Results:" << std::endl;
    std::cout << "  Total runtime:     " << duration << " ms" << std::endl;
    std::cout << "  Archive size:      " << archive.size() << " solutions" << std::endl;
    
    if ((accepted + rejected) > 0) {
        std::cout << "  Acceptance rate:   " << std::fixed << std::setprecision(2)
              << (100.0 * accepted / (accepted + rejected)) << "%" << std::endl;
    } else {
        std::cout << "  Acceptance rate:   N/A" << std::endl;
    }

    if ((feasible + infeasible) > 0) {
        std::cout << "  Feasibility rate:  " << std::fixed << std::setprecision(2)
              << (100.0 * feasible / (feasible + infeasible)) << "%" << std::endl;
    } else {
        std::cout << "  Feasibility rate:  N/A" << std::endl;
    }

    std::cout << std::string(80, '=') << std::endl;
}

# ALNS Improvements for MO-EVRP

## 📋 Tổng Quan Cải Tiến

Phiên bản nâng cấp này cải thiện đáng kể khả năng giải bài toán **Multi-Objective Electric Vehicle Routing Problem (MO-EVRP)** thông qua các cải tiến về:

1. **Local Search nâng cao**
2. **Initial Solution tốt hơn** 
3. **Station Optimization**
4. **Feasibility Check chính xác**
5. **Route Operations cải thiện**

---

## 🚀 Các Cải Tiến Chính

### 1. Enhanced Local Search (4 Operators)

#### a) **Inter-Route Relocate**
```cpp
bool applyRelocate(Solution& solution)
```
- Di chuyển khách hàng từ route này sang route khác
- Tính toán chính xác cost delta (removal savings - insertion cost)
- First improvement strategy để tăng tốc độ

**Ưu điểm:**
- Giảm số lượng routes
- Cân bằng tải giữa các routes
- Tối ưu khoảng cách tổng thể

#### b) **Inter-Route Exchange (2-opt*)**
```cpp
bool applyExchange(Solution& solution)
```
- Hoán đổi khách hàng giữa 2 routes
- Kiểm tra ràng buộc capacity trước khi swap
- Tính cost improvement trước khi thực hiện

**Ưu điểm:**
- Cải thiện cấu trúc routes
- Tối ưu phân bố khách hàng
- Giảm crossing routes

#### c) **Intra-Route 2-opt**
```cpp
bool applyIntraRouteOptimization(Solution& solution)
```
- Đảo ngược segment trong một route
- Loại bỏ crossing edges
- Tối ưu thứ tự viếng thăm

**Ưu điểm:**
- Giảm khoảng cách route
- Loại bỏ detours
- Cải thiện time windows

#### d) **Station Optimization**
```cpp
bool optimizeStations(Solution& solution)
```
- **Remove Unnecessary Stations**: Xóa các trạm sạc không cần thiết
- **Add Necessary Stations**: Thêm trạm khi pin yếu
- Tìm trạm gần nhất để chèn

**Ưu điểm:**
- Giảm số lượng trạm sạc (giảm thời gian)
- Đảm bảo đủ năng lượng
- Tối ưu vị trí sạc

---

### 2. Improved Initial Solution (Clarke-Wright Savings)

```cpp
Solution generateInitialSolution()
```

**Thay vì:** 1 route per customer (tệ)  
**Bây giờ:** Clarke-Wright Savings Algorithm

**Cách hoạt động:**
1. Tạo initial routes (mỗi customer một route)
2. Tính savings cho mọi cặp customers:
   ```
   S(i,j) = d(depot,i) + d(depot,j) - d(i,j)
   ```
3. Sắp xếp savings giảm dần
4. Merge routes theo thứ tự savings

**Ưu điểm:**
- Initial solution tốt hơn nhiều (ít routes hơn)
- ALNS bắt đầu từ baseline tốt
- Tăng tốc độ hội tụ

---

### 3. Station Management

#### a) **Find Nearest Station**
```cpp
std::shared_ptr<Station> findNearestStation(std::shared_ptr<Node> from)
```
- Tìm trạm sạc gần nhất từ một điểm
- Sử dụng khi cần thêm trạm

#### b) **Remove Unnecessary Stations**
```cpp
bool removeUnnecessaryStations(Route& route)
```
- Thử xóa từng trạm sạc
- Kiểm tra feasibility sau khi xóa
- Chỉ xóa nếu route vẫn khả thi

#### c) **Add Necessary Stations**
```cpp
bool addNecessaryStations(Route& route)
```
- Phát hiện vị trí pin yếu (< 10.0)
- Tự động chèn trạm gần nhất
- Đảm bảo năng lượng đủ

---

### 4. Enhanced Feasibility Check

```cpp
bool isFeasible(const Solution& solution)
```

**Kiểm tra đầy đủ:**
- ✅ Route structure (depot at start/end)
- ✅ Battery constraints (không âm, không vượt capacity)
- ✅ Load constraints (không vượt capacity)
- ✅ Time windows (arrival ≤ due date)
- ✅ Customer uniqueness (mỗi customer chỉ phục vụ 1 lần)
- ✅ Completeness (tất cả customers được phục vụ)

**Loại bỏ lỗi:**
- ❌ Không kiểm tra `departure_load == 0` (sai logic)
- ✅ Kiểm tra tổng số customers đúng

---

### 5. Advanced Route Operations

#### Helper Methods mới:

**canSwap()**: Kiểm tra có thể hoán đổi 2 customers không
- Kiểm tra capacity constraints
- Tính toán load mới cho cả 2 routes

**calculateSwapCost()**: Tính cost delta của việc swap
- So sánh old cost vs new cost
- Chính xác đến từng edge

**performSwap()**: Thực hiện swap và recalculate
- Swap nodes
- Recalculate cả 2 routes từ đầu

**canReverse()** & **reverseSegment()**: 
- Đảo ngược một đoạn trong route
- Kiểm tra feasibility
