### 1.1. Tách biệt Validation Logic

**Vấn đề hiện tại:** 
- `Route::canInsert` xử lý quá nhiều logic (capacity, time, battery)
- Khó maintain và test riêng lẻ

**Giải pháp:**

```cpp
// include/validators/RouteValidator.h
#pragma once
#include "Route.h"

class IRouteValidator {
public:
    virtual ~IRouteValidator() = default;
    virtual bool validate(const Route& route, const Node* node, int position) const = 0;
    virtual std::string getErrorMessage() const = 0;
};

class CapacityValidator : public IRouteValidator {
private:
    double maxCapacity;
    std::string errorMsg;
    
public:
    explicit CapacityValidator(double capacity) : maxCapacity(capacity) {}
    
    bool validate(const Route& route, const Node* node, int position) const override {
        if (auto* customer = dynamic_cast<const Customer*>(node)) {
            double totalLoad = route.getCurrentLoad() + customer->demand;
            if (totalLoad > maxCapacity) {
                errorMsg = "Capacity exceeded: " + std::to_string(totalLoad) + " > " + std::to_string(maxCapacity);
                return false;
            }
        }
        return true;
    }
    
    std::string getErrorMessage() const override { return errorMsg; }
};

class TimeWindowValidator : public IRouteValidator {
private:
    mutable std::string errorMsg;
    
public:
    bool validate(const Route& route, const Node* node, int position) const override {
        // Tính toán thời gian đến node mới
        double arrivalTime = route.calculateArrivalTime(node, position);
        
        if (auto* customer = dynamic_cast<const Customer*>(node)) {
            if (arrivalTime > customer->dueDate) {
                errorMsg = "Time window violated: arrival=" + std::to_string(arrivalTime) 
                         + " > dueDate=" + std::to_string(customer->dueDate);
                return false;
            }
        }
        return true;
    }
    
    std::string getErrorMessage() const override { return errorMsg; }
};

class BatteryValidator : public IRouteValidator {
private:
    double maxBattery;
    double energyRate;
    mutable std::string errorMsg;
    
public:
    BatteryValidator(double battery, double rate) 
        : maxBattery(battery), energyRate(rate) {}
    
    bool validate(const Route& route, const Node* node, int position) const override {
        double requiredEnergy = route.calculateEnergyToNode(node, position);
        double availableEnergy = route.getCurrentBattery();
        
        if (requiredEnergy > availableEnergy) {
            errorMsg = "Insufficient battery: required=" + std::to_string(requiredEnergy)
                     + " > available=" + std::to_string(availableEnergy);
            return false;
        }
        return true;
    }
    
    std::string getErrorMessage() const override { return errorMsg; }
};

class CompositeValidator : public IRouteValidator {
private:
    std::vector<std::unique_ptr<IRouteValidator>> validators;
    mutable std::string errorMsg;
    
public:
    void addValidator(std::unique_ptr<IRouteValidator> validator) {
        validators.push_back(std::move(validator));
    }
    
    bool validate(const Route& route, const Node* node, int position) const override {
        for (const auto& validator : validators) {
            if (!validator->validate(route, node, position)) {
                errorMsg = validator->getErrorMessage();
                return false;
            }
        }
        return true;
    }
    
    std::string getErrorMessage() const override { return errorMsg; }
};
```

**Sử dụng:**

```cpp
// Trong Route.cpp

    auto validator = std::make_unique<CompositeValidator>();
    validator->addValidator(std::make_unique<CapacityValidator>(instance->vehicleCapacity));
    validator->addValidator(std::make_unique<TimeWindowValidator>());
    validator->addValidator(std::make_unique<BatteryValidator>(instance->vehicleBattery, instance->vehicleEnergyRate));
    
    return validator->validate(*this, node, position);
}
```

```cpp
// include/core/ServiceLocator.h
#pragma once
#include <memory>
#include <unordered_map>
#include <typeindex>

class ServiceLocator {
private:
    static std::unordered_map<std::type_index, std::shared_ptr<void>> services;
    
public:
    template<typename T>
    static void registerService(std::shared_ptr<T> service) {
        services[std::type_index(typeid(T))] = service;
    }
    
    template<typename T>
    static std::shared_ptr<T> getService() {
        auto it = services.find(std::type_index(typeid(T)));
        if (it != services.end()) {
            return std::static_pointer_cast<T>(it->second);
        }
        return nullptr;
    }
    
    static void clear() {
        services.clear();
    }
};
```

```cpp
// include/objectives/ObjectiveFunction.h
#pragma once
#include "Solution.h"

class IObjective {
public:
    virtual ~IObjective() = default;
    virtual double evaluate(const Solution& solution) const = 0;
    virtual std::string getName() const = 0;
    virtual bool minimize() const = 0; // true = minimize, false = maximize
};

class NumberOfVehiclesObjective : public IObjective {
public:
    double evaluate(const Solution& solution) const override {
        return solution.getRoutes().size();
    }
    
    std::string getName() const override { return "Number of Vehicles"; }
    bool minimize() const override { return true; }
};

class TotalDistanceObjective : public IObjective {
public:
    double evaluate(const Solution& solution) const override {
        double total = 0.0;
        for (const auto& route : solution.getRoutes()) {
            total += route.getTotalDistance();
        }
        return total;
    }
    
    std::string getName() const override { return "Total Distance"; }
    bool minimize() const override { return true; }
};

class TotalEnergyObjective : public IObjective {
public:
    double evaluate(const Solution& solution) const override {
        double total = 0.0;
        for (const auto& route : solution.getRoutes()) {
            total += route.getTotalEnergyCharged();
        }
        return total;
    }
    
    std::string getName() const override { return "Total Energy Cost"; }
    bool minimize() const override { return true; }
};

class MakespanObjective : public IObjective {
public:
    double evaluate(const Solution& solution) const override {
        double maxTime = 0.0;
        for (const auto& route : solution.getRoutes()) {
            maxTime = std::max(maxTime, route.getTotalTime());
        }
        return maxTime;
    }
    
    std::string getName() const override { return "Makespan"; }
    bool minimize() const override { return true; }
};

class ObjectiveManager {
private:
    std::vector<std::shared_ptr<IObjective>> objectives;
    
public:
    void addObjective(std::shared_ptr<IObjective> obj) {
        objectives.push_back(obj);
    }
    
    std::vector<double> evaluateAll(const Solution& solution) const {
        std::vector<double> values;
        for (const auto& obj : objectives) {
            values.push_back(obj->evaluate(solution));
        }
        return values;
    }
    
    size_t getObjectiveCount() const { return objectives.size(); }
    
    const IObjective* getObjective(size_t index) const {
        return index < objectives.size() ? objectives[index].get() : nullptr;
    }
};
```

### 2.1. Distance Matrix Caching

```cpp
// include/core/DistanceMatrix.h
#pragma once
#include <vector>
#include <memory>
#include "Node.h"

class DistanceMatrix {
private:
    std::vector<std::vector<double>> distances;
    std::vector<std::vector<double>> times;
    size_t size;
    double velocity;
    
public:
    DistanceMatrix(const std::vector<Node*>& nodes, double vehicleVelocity) 
        : size(nodes.size()), velocity(vehicleVelocity) {
        
        // Pre-allocate
        distances.resize(size, std::vector<double>(size, 0.0));
        times.resize(size, std::vector<double>(size, 0.0));
        
        // Precompute all distances and times
        for (size_t i = 0; i < size; ++i) {
            for (size_t j = 0; j < size; ++j) {
                if (i != j) {
                    double dist = Utils::euclideanDistance(
                        nodes[i]->x, nodes[i]->y,
                        nodes[j]->x, nodes[j]->y
                    );
                    distances[i][j] = dist;
                    times[i][j] = dist / velocity;
                }
            }
        }
    }
    
    double getDistance(int from, int to) const {
        return distances[from][to];
    }
    
    double getTime(int from, int to) const {
        return times[from][to];
    }
    
    // Cache-friendly batch lookup
    std::vector<double> getDistancesFrom(int from) const {
        return distances[from];
    }
};
```

### 2.2. Memory Pool cho Route Objects

```cpp
// include/utils/ObjectPool.h
#pragma once
#include <vector>
#include <memory>
#include <mutex>

template<typename T>
class ObjectPool {
private:
    std::vector<std::unique_ptr<T>> pool;
    std::vector<T*> available;
    std::mutex mtx;
    size_t initialSize;
    
    void grow(size_t count = 10) {
        for (size_t i = 0; i < count; ++i) {
            auto obj = std::make_unique<T>();
            available.push_back(obj.get());
            pool.push_back(std::move(obj));
        }
    }
    
public:
    explicit ObjectPool(size_t initial = 50) : initialSize(initial) {
        grow(initial);
    }
    
    T* acquire() {
        std::lock_guard<std::mutex> lock(mtx);
        
        if (available.empty()) {
            grow();
        }
        
        T* obj = available.back();
        available.pop_back();
        obj->reset(); // Reset to default state
        return obj;
    }
    
    void release(T* obj) {
        std::lock_guard<std::mutex> lock(mtx);
        available.push_back(obj);
    }
    
    size_t poolSize() const { return pool.size(); }
    size_t availableCount() const { return available.size(); }
};

// Route cần có method reset()
class Route {
public:
    void reset() {
        nodes.clear();
        // Reset all internal states
    }
};
```

```cpp
};
```

### 2.3. Parallel Evaluation

```cpp
// include/solvers/ParallelEvaluator.h
#pragma once
#include <vector>
#include <future>
#include <thread>
#include "Solution.h"
#include "ObjectiveFunction.h"

class ParallelEvaluator {
private:
    ObjectiveManager& objectives;
    size_t numThreads;
    
public:
    ParallelEvaluator(ObjectiveManager& obj, size_t threads = 0) 
        : objectives(obj), numThreads(threads == 0 ? std::thread::hardware_concurrency() : threads) {}
    
    void evaluateBatch(std::vector<Solution*>& solutions) {
        size_t batchSize = solutions.size();
        size_t chunkSize = (batchSize + numThreads - 1) / numThreads;
        
        std::vector<std::future<void>> futures;
        
        for (size_t t = 0; t < numThreads; ++t) {
            size_t start = t * chunkSize;
            size_t end = std::min(start + chunkSize, batchSize);
            
            if (start >= batchSize) break;
            
            futures.push_back(std::async(std::launch::async, [this, &solutions, start, end]() {
                for (size_t i = start; i < end; ++i) {
                    auto values = objectives.evaluateAll(*solutions[i]);
                    solutions[i]->setObjectiveValues(values);
                }
            }));
        }
        
        // Wait for all threads
        for (auto& f : futures) {
            f.wait();
        }
    }
};
```

## 3. Nâng cấp Thuật toán ALNS

### 3.1. Thêm Destroy Operators

```cpp
// include/operators/DestroyOperators.h
#pragma once
#include "Solution.h"
#include <random>

class IDestroyOperator {
public:
    virtual ~IDestroyOperator() = default;
    virtual std::vector<Customer*> destroy(Solution& solution, double destructionRate, std::mt19937& rng) = 0;
    virtual std::string getName() const = 0;
};

// 1. Random Removal
class RandomRemoval : public IDestroyOperator {
public:
    std::vector<Customer*> destroy(Solution& solution, double rate, std::mt19937& rng) override {
        auto& routes = solution.getRoutes();
        std::vector<Customer*> removed;
        
        int totalCustomers = solution.getCustomerCount();
        int toRemove = static_cast<int>(totalCustomers * rate);
        
        std::uniform_int_distribution<> routeDist(0, routes.size() - 1);
        
        while (removed.size() < toRemove && solution.getCustomerCount() > 0) {
            int routeIdx = routeDist(rng);
            auto* route = routes[routeIdx];
            
            if (route->getCustomerCount() > 0) {
                std::uniform_int_distribution<> nodeDist(1, route->size() - 2); // Exclude depot
                int nodeIdx = nodeDist(rng);
                
                auto* customer = dynamic_cast<Customer*>(route->getNode(nodeIdx));
                if (customer) {
                    removed.push_back(customer);
                    route->removeNode(nodeIdx);
                }
            }
        }
        
        return removed;
    }
    
    std::string getName() const override { return "Random Removal"; }
};

// 2. Worst Distance Removal
class WorstDistanceRemoval : public IDestroyOperator {
public:
    std::vector<Customer*> destroy(Solution& solution, double rate, std::mt19937& rng) override {
        auto& routes = solution.getRoutes();
        std::vector<Customer*> removed;
        
        int toRemove = static_cast<int>(solution.getCustomerCount() * rate);
        
        // Calculate savings for each customer
        struct CustomerSaving {
            Customer* customer;
            Route* route;
            int position;
            double saving;
        };
        
        std::vector<CustomerSaving> savings;
        
        for (auto* route : routes) {
            for (int i = 1; i < route->size() - 1; ++i) {
                auto* customer = dynamic_cast<Customer*>(route->getNode(i));
                if (!customer) continue;
                
                // Calculate removal saving
                int prev = route->getNode(i-1)->id;
                int curr = customer->id;
                int next = route->getNode(i+1)->id;
                
                auto instance = ServiceLocator::getService<Instance>();
                double currentDist = instance->getDistance(prev, curr) + instance->getDistance(curr, next);
                double newDist = instance->getDistance(prev, next);
                double saving = currentDist - newDist;
                
                savings.push_back({customer, route, i, saving});
            }
        }
        
        // Sort by saving (worst = highest distance)
        std::sort(savings.begin(), savings.end(), [](const auto& a, const auto& b) {
            return a.saving > b.saving;
        });
        
        // Remove worst customers
        for (int i = 0; i < toRemove && i < savings.size(); ++i) {
            removed.push_back(savings[i].customer);
            savings[i].route->removeNode(savings[i].position);
        }
        
        return removed;
    }
    
    std::string getName() const override { return "Worst Distance Removal"; }
};

// 3. Shaw Removal (relatedness-based)
class ShawRemoval : public IDestroyOperator {
private:
    double relatedness(const Customer* c1, const Customer* c2) const {
        auto instance = ServiceLocator::getService<Instance>();
        
        // Distance component
        double dist = instance->getDistance(c1->id, c2->id);
        double maxDist = 1000.0; // Normalize
        double distWeight = 0.5;
        
        // Time window component
        double timeDiff = std::abs(c1->readyTime - c2->readyTime);
        double maxTimeDiff = 100.0;
        double timeWeight = 0.3;
        
        // Demand component
        double demandDiff = std::abs(c1->demand - c2->demand);
        double maxDemandDiff = 50.0;
        double demandWeight = 0.2;
        
        double score = distWeight * (dist / maxDist) 
                     + timeWeight * (timeDiff / maxTimeDiff)
                     + demandWeight * (demandDiff / maxDemandDiff);
        
        return score;
    }
    
public:
    std::vector<Customer*> destroy(Solution& solution, double rate, std::mt19937& rng) override {
        auto& routes = solution.getRoutes();
        std::vector<Customer*> removed;
        
        int toRemove = static_cast<int>(solution.getCustomerCount() * rate);
        if (toRemove == 0) return removed;
        
        // Select seed customer randomly
        std::uniform_int_distribution<> routeDist(0, routes.size() - 1);
        Route* seedRoute = routes[routeDist(rng)];
        
        if (seedRoute->getCustomerCount() == 0) return removed;
        
        std::uniform_int_distribution<> nodeDist(1, seedRoute->size() - 2);
        auto* seedCustomer = dynamic_cast<Customer*>(seedRoute->getNode(nodeDist(rng)));
        
        if (!seedCustomer) return removed;
        
        // Calculate relatedness to seed
        std::vector<std::pair<Customer*, double>> candidates;
        
        for (auto* route : routes) {
            for (int i = 1; i < route->size() - 1; ++i) {
                auto* customer = dynamic_cast<Customer*>(route->getNode(i));
                if (customer && customer != seedCustomer) {
                    double rel = relatedness(seedCustomer, customer);
                    candidates.push_back({customer, rel});
                }
            }
        }
        
        // Sort by relatedness
        std::sort(candidates.begin(), candidates.end(), [](const auto& a, const auto& b) {
            return a.second < b.second;
        });
        
        // Remove most related customers
        removed.push_back(seedCustomer);
        seedRoute->removeCustomer(seedCustomer);
        
        for (int i = 0; i < toRemove - 1 && i < candidates.size(); ++i) {
            removed.push_back(candidates[i].first);
            // Find and remove from route
            for (auto* route : routes) {
                if (route->removeCustomer(candidates[i].first)) {
                    break;
                }
            }
        }
        
        return removed;
    }
    
    std::string getName() const override { return "Shaw Removal"; }
};

// 4. Time-based Removal
class TimeWindowRemoval : public IDestroyOperator {
public:
    std::vector<Customer*> destroy(Solution& solution, double rate, std::mt19937& rng) override {
        auto& routes = solution.getRoutes();
        std::vector<Customer*> removed;
        
        int toRemove = static_cast<int>(solution.getCustomerCount() * rate);
        
        // Find customers with tight time windows or violations
        struct CustomerSlack {
            Customer* customer;
            Route* route;
            int position;
            double slack;
        };
        
        std::vector<CustomerSlack> candidates;
        
        for (auto* route : routes) {
            for (int i = 1; i < route->size() - 1; ++i) {
                auto* customer = dynamic_cast<Customer*>(route->getNode(i));
                if (!customer) continue;
                
                double arrivalTime = route->getArrivalTime(i);
                double slack = customer->dueDate - arrivalTime;
                
                candidates.push_back({customer, route, i, slack});
            }
        }
        
        // Sort by slack (smallest = tightest)
        std::sort(candidates.begin(), candidates.end(), [](const auto& a, const auto& b) {
            return a.slack < b.slack;
        });
        
        // Remove customers with tightest windows
        for (int i = 0; i < toRemove && i < candidates.size(); ++i) {
            removed.push_back(candidates[i].customer);
            candidates[i].route->removeNode(candidates[i].position);
        }
        
        return removed;
    }
    
    std::string getName() const override { return "Time Window Removal"; }
};

// 5. Station-related Removal
class StationRelatedRemoval : public IDestroyOperator {
public:
    std::vector<Customer*> destroy(Solution& solution, double rate, std::mt19937& rng) override {
        auto& routes = solution.getRoutes();
        std::vector<Customer*> removed;
        
        int toRemove = static_cast<int>(solution.getCustomerCount() * rate);
        
        // Find customers near charging stations
        auto instance = ServiceLocator::getService<Instance>();
        
        for (auto* route : routes) {
            for (int i = 1; i < route->size() - 1 && removed.size() < toRemove; ++i) {
                auto* node = route->getNode(i);
                
                // If it's a station, remove nearby customers
                if (dynamic_cast<Station*>(node)) {
                    // Remove previous customer
                    if (i > 1) {
                        auto* prevCustomer = dynamic_cast<Customer*>(route->getNode(i-1));
                        if (prevCustomer) {
                            removed.push_back(prevCustomer);
                            route->removeNode(i-1);
                            --i;
                        }
                    }
                    
                    // Remove next customer
                    if (i < route->size() - 2) {
                        auto* nextCustomer = dynamic_cast<Customer*>(route->getNode(i+1));
                        if (nextCustomer) {
                            removed.push_back(nextCustomer);
                            route->removeNode(i+1);
                        }
                    }
                }
            }
        }
        
        return removed;
    }
    
    std::string getName() const override { return "Station-related Removal"; }
};
```

// 1. Greedy Insertion

        auto& routes = solution.getRoutes();

            double bestCost = std::numeric_limits<double>::max();
            Customer* bestCustomer = nullptr;
            Route* bestRoute = nullptr;
            int bestPosition = -1;

                for (auto* route : routes) {
                    for (int pos = 1; pos < route->size(); ++pos) {
                        if (route->canInsert(customer, pos)) {
                            double cost = route->getInsertionCost(customer, pos);
